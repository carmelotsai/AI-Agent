      ? `上一份日誌（${previous.day}）還沒完成的事如下。本次對話中沒有被完成或取消的，要延續到這份日誌的「還沒完成的事」；` +
        `被完成或取消的就拿掉，並在「今天做了什麼」註明：\n\n${carried}`
      : "",
    "格式如下，四個章節標題固定、不可增刪；沒有內容的章節寫「- （無）」：",
    [`# ${day} 工作日誌`, ...JOURNAL_HEADINGS].join("\n\n"),
    "規則：只寫對話裡真的發生的事，不要推測；「產出的文件」只列真的存檔的檔案路徑；" +
      "「還沒完成的事」要列出對話中提到、但還沒完成的待辦、要交的東西、要提醒的事，有負責人或期限就一起寫；" +

  ].filter(Boolean).join("\n\n");
}

let replyBuffer = "";
let lastAgentError = "";
function onAgentEvent(event) {
  if (event.type === "message_update" && event.assistantMessageEvent.type === "text_delta") {
    replyBuffer += event.assistantMessageEvent.delta;
  }
  if (event.type === "message_end" && event.message.role === "assistant" && event.message.errorMessage) {
    lastAgentError = event.message.errorMessage;
  }
  if (event.type === "auto_retry_end" && !event.success && event.finalError) {
    lastAgentError = event.finalError;
  }
}

// ── 外掛：一個功能一個資料夾（外掛/名稱/index.mjs）；拿掉資料夾，那個功能就消失 ──
const PLUGIN_DIR = join(ROOT, "外掛");
/** 外掛可以用的 bridge 功能。 */
const pluginApi = {
  root: ROOT,
  /** 把檔案傳給正在跟秘書說話的使用者；失敗會丟出錯誤。 */
  async sendFile(path) {
    if (!running) throw new Error("現在沒有在處理使用者的訊息，無法傳檔案");
    await sendDocument(running.chatId, path);
  },
  /** 在這則回覆最後加一行說明（同一個 key 只留最後一次），例如外掛這次真的查了哪些資料。 */
  setReplyNote(key, text) {
    running?.notes.set(key, text);
  },
};

/** 讀取 外掛/ 底下每個資料夾的 index.mjs；載入失敗的外掛略過，不影響其他功能。 */
async function loadPlugins() {
  if (!existsSync(PLUGIN_DIR)) return [];
  const loaded = [];
  for (const entry of readdirSync(PLUGIN_DIR, { withFileTypes: true })) {
    const file = join(PLUGIN_DIR, entry.name, "index.mjs");
    if (!entry.isDirectory() || !existsSync(file)) continue;
    try {
      const module = await import(pathToFileURL(file).href);
      if (typeof module.default !== "function") throw new Error("index.mjs 要用 export default 匯出一個函式");
      loaded.push({
        name: entry.name,
        prompt: module.prompt ?? "",
        levels: module.levels ?? {}, // 外掛宣告自己的工具屬於哪一級；沒宣告的算「執行」
        extension: { name: entry.name, factory: (pi) => module.default(pi, pluginApi) },
      });
    } catch (error) {
      console.error(`外掛「${entry.name}」載入失敗，先略過：`, errorText(error));
    }
  }
  return loaded;
}
const plugins = await loadPlugins();

// ── 權限分級：查閱直接做、擬稿限定位置、執行先問使用者（規則在 權限.json）──────
const PERMISSION_FILE = join(ROOT, "權限.json");
const DEFAULT_PERMISSIONS = {
  可以寫的地方: ["文件/", ".pi/skills/"],
  只能修改的檔案: ["MEMORY.md"],
  不能讀的地方: [".env", "data/"],
  等我確認幾分鐘: 5,
};
const LEVELS = { read: "查閱", grep: "查閱", find: "查閱", ls: "查閱", write: "擬稿", edit: "擬稿" };

/** 讀 權限.json；沒有檔案或格式錯誤就用預設值。 */
function loadPermissions() {
  if (!existsSync(PERMISSION_FILE)) return DEFAULT_PERMISSIONS;
  try {
    const custom = JSON.parse(readFileSync(PERMISSION_FILE, "utf8"));
    const merged = { ...DEFAULT_PERMISSIONS, ...custom };
    const lists = ["可以寫的地方", "只能修改的檔案", "不能讀的地方"];
    if (!lists.every((key) => Array.isArray(merged[key]) && merged[key].every((item) => typeof item === "string")) ||
      !(merged.等我確認幾分鐘 > 0)) {
      throw new Error("欄位格式不對");
    }
    return merged;
  } catch (error) {
    console.error("權限.json 讀不懂，先用預設權限：", errorText(error));
    return DEFAULT_PERMISSIONS;
  }
}
const permissions = loadPermissions();

/** 工具屬於哪一級：內建工具固定；外掛工具看外掛宣告，沒宣告的算「執行」。 */
function levelOf(toolName) {
  return LEVELS[toolName] ?? plugins.map((plugin) => plugin.levels[toolName]).find(Boolean) ?? "執行";
}

/** 路徑換成相對於專案資料夾、用 / 分隔；在專案外面的回傳 null。 */
function projectPath(path) {
  const rel = relative(ROOT, resolve(ROOT, path ?? ".")).split(sep).join("/");
  return rel.startsWith("../") || rel === ".." || /^[a-zA-Z]:/.test(rel) ? null : rel;
}
const inside = (rel, places) => places.some((place) => place.endsWith("/") ? `${rel}/`.startsWith(place) : rel === place);

function permissionPrompt() {
  return [
    "你的工具分三級，由系統在執行前檢查：",
    `查閱（read、grep、find、ls）：直接做，但只能看本專案資料夾，不能讀 ${permissions.不能讀的地方.join("、")}。`,
    `擬稿（write、edit）：只能寫 ${permissions.可以寫的地方.join("、")}；${permissions.只能修改的檔案.join("、")} 只能用 edit 修改，不能整份覆寫。`,
    "執行（" + SHELL_TOOL + " 與部分外掛工具）：系統會先傳按鈕問使用者，使用者按了「執行」才會做。",
    "工具被擋下或使用者沒有同意時，照實告訴使用者原因；不要改用其他工具或其他寫法繞過。",
  ].join("\n");
}

/**
 * 執行級指令的警示：提到秘書平常不能改的檔案、或是刪除類指令，就在按鈕上方加一行提醒。
 * 直接比對指令文字，不靠模型判斷；換個寫法可能比對不到，所以只是輔助，決定權仍在使用者。
 */
function warningsFor(command) {
  const writable = permissions.可以寫的地方.map((place) => place.split("/")[0]);
  const escape = (text) => text.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  const protectedNames = readdirSync(ROOT)
    .filter((name) => name !== "node_modules" && !writable.includes(name))
    .filter((name) => new RegExp(`(^|[\\\\/"'\\s])${escape(name)}($|[\\\\/"'\\s;,)])`, "i").test(command));
  const warnings = [];
  if (protectedNames.length > 0) warnings.push(`⚠️ 會碰到秘書平常不能改的檔案：${protectedNames.join("、")}`);
  if (/(^|[\s;|&(])(Remove-Item|rm|rmdir|rd|del|erase|ri)(\s|$)/i.test(command)) warnings.push("⚠️ 這個指令會刪除東西");
  return warnings;
}

/** 檢查一次工具呼叫；回傳擋下的原因（寫成秘書可以直接轉告使用者的話），可以執行就回傳 undefined。 */
async function checkPermission(event) {
  const level = levelOf(event.toolName);
  if (level === "查閱") {
    const rel = projectPath(event.input.path);
    if (rel === null) return `我只能看專案資料夾裡的東西，沒有權限看 ${event.input.path}。`;
    if (inside(rel, permissions.不能讀的地方)) return `我沒有權限讀 ${rel}。`;
    return undefined;
  }
  if (level === "擬稿") {
    const rel = projectPath(event.input.path);
    if (rel !== null && inside(rel, permissions.可以寫的地方)) return undefined;
    if (rel !== null && event.toolName === "edit" && inside(rel, permissions.只能修改的檔案)) return undefined;
    return `我沒有權限${event.toolName === "write" ? "寫入" : "修改"} ${rel ?? event.input.path}：我只能寫 ${permissions.可以寫的地方.join("、")}，` +
      `${permissions.只能修改的檔案.join("、")} 只能修改、不能整份覆寫。`;
  }
  const command = event.toolName === SHELL_TOOL ? event.input.command : JSON.stringify(event.input);
  const warnings = event.toolName === SHELL_TOOL ? warningsFor(command) : [];
  const answer = await askUser([`秘書要執行：${event.toolName}`, ...warnings, command.slice(0, 3000)].join("\n\n"));
  if (answer === "run") return undefined;
  return {
    skip: "你按了「不要」，所以我沒有執行。",
    timeout: `你 ${permissions.等我確認幾分鐘} 分鐘內沒有按按鈕，所以我沒有執行。`,
    stop: "你用 /stop 叫停了，所以我沒有執行。",
    none: "現在沒有人在跟我說話，所以我沒有執行。",
    failed: "確認訊息傳不出去，所以我沒有執行。",
  }[answer];
}

/** 內建外掛：每次呼叫工具前先檢查權限（不在 外掛/ 裡，拿不掉）。 */
const permissionExtension = {
  name: "權限分級",
  hidden: true,
  factory: (pi) => pi.on("tool_call", async (event) => {
    const reason = await checkPermission(event);
    if (!reason) return undefined;
    console.log(`擋下 ${event.toolName}：${reason}`);
    return { block: true, reason: `${reason}（請照實轉告使用者，不要改用其他工具或其他寫法繞過。）` };
  }),
};

let pendingAsk = null; // 等使用者按按鈕的那一題 { resolve, chatId, messageId, text }
const ANSWER_NOTES = { run: "✅ 已同意執行", skip: "❌ 你按了不要", timeout: "⌛ 太久沒回應，沒有執行", stop: "⏹ 已用 /stop 取消" };

/** 傳一則附「執行／不要」按鈕的訊息，等使用者按；回傳 run、skip、timeout、stop、none 或 failed。 */
async function askUser(text) {
  if (!running) return "none";
  const chatId = running.chatId;
  let sent;
  try {
    sent = await tg("sendMessage", {
      chat_id: chatId,
      text,
      reply_markup: { inline_keyboard: [[{ text: "✅ 執行", callback_data: "run" }, { text: "❌ 不要", callback_data: "skip" }]] },
    });
  } catch (error) {
    console.error("確認訊息傳不出去：", errorText(error));
    return "failed";
  }
  const answer = await new Promise((resolve) => {
    const timer = setTimeout(() => resolve("timeout"), permissions.等我確認幾分鐘 * 60_000);
    pendingAsk = { resolve: (value) => { clearTimeout(timer); resolve(value); }, chatId, messageId: sent.result.message_id, text };
  });
  const { messageId } = pendingAsk;
  pendingAsk = null;
  // 把按鈕換成結果，舊按鈕就不能再按
  await tg("editMessageText", { chat_id: chatId, message_id: messageId, text: `${text}\n\n${ANSWER_NOTES[answer]}` })
    .catch((error) => console.error("更新按鈕訊息失敗：", errorText(error)));
  return answer;
}

/** 使用者按了按鈕。 */
async function onButton(query) {
  const matches = pendingAsk && query.message?.message_id === pendingAsk.messageId;
  await tg("answerCallbackQuery", { callback_query_id: query.id, ...(matches ? {} : { text: "這個按鈕已經失效了。" }) })
    .catch((error) => console.error("回應按鈕失敗：", errorText(error)));
  if (matches && (query.data === "run" || query.data === "skip")) pendingAsk.resolve(query.data);
}

/** 用某一本對話存檔建立 Pi session；技能只讀專案內的 .pi/skills/，外掛只載入 外掛/ 裡的。 */
async function openSession(manager, day) {
  const loader = new DefaultResourceLoader({
    cwd: ROOT,
    agentDir: AGENT_DIR,
    noExtensions: true, // 不自動載入 ~/.pi 等處的 extensions，只用下面交給它的外掛
    extensionFactories: [permissionExtension, ...plugins.map((plugin) => plugin.extension)],
    noPromptTemplates: true,
    noThemes: true,
    noContextFiles: true,
    systemPromptOverride: () => buildSystemPrompt(day, toolNamesOf(loader)),
  });
  await loader.reload();
  for (const { path, error } of loader.getExtensions().errors) console.error(`外掛 ${path} 啟動失敗：`, error);
  const { session } = await createAgentSession({
    cwd: ROOT,
    agentDir: AGENT_DIR,
    modelRuntime,
    model,
    resourceLoader: loader,
    tools: toolNamesOf(loader),
    sessionManager: manager,
    thinkingLevel: THINKING_LEVEL,
  });
  session.subscribe(onAgentEvent);
  return { session, manager, day, skills: loader.getSkills().skills.map((skill) => skill.name) };
}

/** 對話存檔檔名是「時間戳_工作日-序號.jsonl」，依日期、序號排序。 */
function sessionFiles() {
  return readdirSync(SESSION_DIR)
    .map((name) => {
      const match = name.match(/_(\d{4}-\d{2}-\d{2})-(\d+)\.jsonl$/);
      return match && { path: join(SESSION_DIR, name), day: match[1], seq: Number(match[2]) };
    })
    .filter(Boolean)
    .sort((a, b) => a.day.localeCompare(b.day) || a.seq - b.seq);
}

function isJournaled(manager) {
  return manager.getEntries().some((entry) => entry.type === "custom" && entry.customType === JOURNAL_DONE);
}

function hasConversation(manager) {
  return manager.getEntries().some((entry) => entry.type === "message" && entry.message.role === "user");
}

function textFromLatestAssistantMessage(session) {
  const message = [...session.messages].reverse().find((item) => item.role === "assistant");
  if (!message || !Array.isArray(message.content)) return "";
  return message.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("");
}

/** 把一本對話整理成 日誌/YYYY-MM-DD.md；同一天已有日誌就合併。由程式寫檔，不靠模型用工具。 */
async function writeJournal(target) {
  const file = `${target.day}.md`;
  const path = join(JOURNAL_DIR, file);
  const earlier = existsSync(path) ? readFileSync(path, "utf8").trim() : "";
  // 同一天稍早的日誌已經延續過待辦，只有這一天第一次整理時才需要帶上一份日誌的待辦。
  const previousFile = earlier ? undefined : journalFiles().filter((name) => name < file).at(-1);
  const previous = previousFile && {
    day: previousFile.replace(/\.md$/, ""),
    text: readFileSync(join(JOURNAL_DIR, previousFile), "utf8"),
  };
  replyBuffer = "";
  lastAgentError = "";
  await promptSession(target.session, journalPrompt(target.day, earlier, previous));
  const text = textFromLatestAssistantMessage(target.session)
    .trim()
    .replace(/^```(?:markdown|md)?\s*\n/, "")
    .replace(/\n```\s*$/, "")
    .trim();
  if (!JOURNAL_HEADINGS.every((heading) => text.includes(heading))) {
    throw new Error(lastAgentError || "日誌格式不完整");
  }
  writeFileSync(path, `${text}\n`, "utf8");
  target.manager.appendCustomEntry(JOURNAL_DONE, { file: `日誌/${file}` });
  return `日誌/${file}`;
}

/** 把還沒整理過的舊對話都寫成日誌（例如昨天關機前沒整理到的）。失敗的下次再試。 */
async function journalPendingSessions(exceptPath) {
  const written = [];
  for (const file of sessionFiles()) {
    if (stopRequested()) break; // 使用者傳了 /stop，剩下的留到下次再補
    if (file.path === exceptPath) continue;
    const manager = SessionManager.open(file.path, SESSION_DIR);
    if (isJournaled(manager) || !hasConversation(manager)) continue;
    const target = await openSession(manager, file.day);
    try {
      written.push(await writeJournal(target));
      console.log(`已整理工作日誌：日誌/${file.day}.md`);
    } catch (error) {
      console.error(`整理 ${file.day} 的工作日誌失敗，下次再試：`, errorText(error));
    } finally {
      target.session.dispose();
    }
  }
  return [...new Set(written)];
}

let current = null;
/** 上班：今天還有沒整理的那本就接著用，否則開新的一本；開之前先把舊的整理成日誌。 */
async function startWorkday() {
  const day = workday();
  const last = sessionFiles().filter((file) => file.day === day).at(-1);
  const lastManager = last && SessionManager.open(last.path, SESSION_DIR);
  const resume = lastManager && !isJournaled(lastManager) ? lastManager : null;
  const written = await journalPendingSessions(resume ? last.path : undefined);
  const manager = resume ?? SessionManager.create(ROOT, SESSION_DIR, { id: `${day}-${(last?.seq ?? 0) + 1}` });
  current = await openSession(manager, day);
  return written;
}

function errorText(error) {
  if (error instanceof Error) return error.message;
  if (typeof error === "string") return error;
  try {
    return JSON.stringify(error);
  } catch {
    return String(error);
  }
}

function readableAgentError(error) {
  const raw = [errorText(error), lastAgentError].filter(Boolean).join("\n").trim();
  if (/quota|rate.?limit|usage|429|too many|capacity|exceed|billing|resource_exhausted/i.test(raw)) {
    return "Gemini 免費額度或速率上限可能已到（Flash-Lite 免費層每分鐘、每天都有上限）。" +
      "系統已自動重試過幾次仍失敗，請稍等一分鐘再試；如果常常發生，建議到 Google AI Studio 開通計費解除限制。";
  }
  return raw
    ? `Agent 暫時無法回覆：${raw.slice(0, 1000)}`
    : "Agent 暫時無法回覆，請稍後再試。";
}

async function chat(text) {
  let notice = "";
  if (current.day !== workday()) {
    current.session.dispose();
    const written = await startWorkday();
    if (written.length > 0) notice = `（新的一天，已整理好工作日誌：${written.join("、")}）\n\n`;
  }
  replyBuffer = "";
  lastAgentError = "";
  if (stopRequested()) return notice;
  const leafBefore = current.manager.getLeafId();
  try {
    await promptSession(current.session, `【現在時間：${nowLabel()}】\n${text}`);
    // 只送最後一則回覆：模型在呼叫工具前寫的規劃文字不給使用者看。
    const reply = (textFromLatestAssistantMessage(current.session).trim() || replyBuffer.trim())
      .replace(/^【現在時間：[^】]*】\s*/, ""); // 模型偶爾會照抄訊息開頭的時間標記
    const notes = [...(running?.notes.values() ?? [])];
    return notice + (reply || readableAgentError()) + (notes.length > 0 ? `\n\n${notes.join("\n")}` : "");
  } catch (error) {
    return notice + readableAgentError(error);
  } finally {
    if (stopRequested()) await forgetStoppedTurn(leafBefore, text);
  }
}

/**
 * 被叫停的那一輪只留下做到一半的工具結果，秘書下一則常會接著回報、甚至說做完了。
 * 所以把對話退回這則訊息之前（原本的紀錄仍留在存檔的分支裡），改記一句「被叫停了」。
 */
async function forgetStoppedTurn(leafBefore, text) {
  try {
    const entries = current.manager.getEntries();
    const index = entries.findLastIndex((entry) => entry.parentId === leafBefore && entry.type === "message");
    const turn = index < 0 ? [] : entries.slice(index).map((entry) => entry.message).filter(Boolean);
    // 列出叫停前做過的動作（只列工具與參數，不放結果內容，免得秘書又去回報）
    const calls = new Map(turn.flatMap((message) => message.role === "assistant" && Array.isArray(message.content)
      ? message.content.filter((block) => block.type === "toolCall").map((block) => [block.id, block])
      : []));
    const done = turn.filter((message) => message.role === "toolResult").map((result) => {
      const output = JSON.stringify(result.content);
      const status = !result.isError ? "完成" : /aborted/i.test(output) ? "被中斷" : /沒有執行|沒有權限/.test(output) ? "沒有執行" : "失敗";
      const args = JSON.stringify(calls.get(result.toolCallId)?.arguments ?? {}).slice(0, 150);
      return `- ${result.toolName} ${args}：${status}`;
    });
    if (index >= 0) await current.session.navigateTree(entries[index].id);
    await current.session.sendCustomMessage({ customType: STOPPED, content: stoppedNote(text, done), display: false });
  } catch (error) {
    console.error("整理被叫停的對話失敗：", errorText(error));
  }
}

/** /new：今天先到這，整理日誌並開新的一本。 */
async function newConversation() {
  if (!hasConversation(current.manager)) return "目前沒有新的對話需要整理。";
  let file;
  try {
    file = await writeJournal(current);
  } catch (error) {

// ── 緊急停止：記下正在做的事與排隊中的訊息，/stop 不排隊、直接叫停 ──────
let running = null;        // 正在處理的訊息 { stopped, chatId, notes }
const waiting = new Set(); // 排隊中的訊息
let activeSession = null;  // 正在呼叫模型的 session（對話本身，或正在整理日誌的那一本）
const stopRequested = () => running?.stopped === true;

/** 呼叫模型，並記下是哪個 session，/stop 才知道要中止誰。 */
async function promptSession(session, text) {
  activeSession = session;
  try {
    await session.prompt(text);
  } finally {
    activeSession = null;
  }
}

let queue = Promise.resolve();
/** 把一則訊息交給秘書，回傳完整文字回覆（被 /stop 叫停的回傳 null）；訊息排隊、一次一則。 */
function handleMessage(text, chatId) {
  const command = text.trim().match(/^\/(new|recap)(?:@\w+)?$/)?.[1];
  const task = { stopped: false, chatId, notes: new Map() }; // chatId：外掛傳檔案給誰；notes：外掛要附在回覆最後的話
  waiting.add(task);
  const run = queue.then(async () => {
    waiting.delete(task);
    if (task.stopped) return null;
    running = task;
    try {
      let reply;
      if (command === "new") reply = await newConversation();
      else if (command === "recap") reply = recap();
      else reply = await chat(text);
      return task.stopped ? null : reply;
    } finally {
      running = null;
    }
  });
  queue = run.catch(() => {});
  return run;
}

/** /stop：立刻叫停正在做的事，並取消排隊中的訊息。 */
async function stop() {
  const cancelled = [...waiting].filter((task) => !task.stopped);
  for (const task of cancelled) task.stopped = true;
  const wasRunning = running !== null;
  if (!wasRunning && cancelled.length === 0) return "目前沒有在做的事。";
  if (wasRunning) {
    running.stopped = true;
    pendingAsk?.resolve("stop"); // 正在等使用者按按鈕的，一起取消
    // 等秘書停下最多 5 秒；就算中止過程出錯，也一定要回覆使用者。
    await Promise.race([
      Promise.resolve(activeSession?.abort()).catch((error) => console.error("中止時發生錯誤：", errorText(error))),
      new Promise((resolve) => setTimeout(resolve, 5000)),
    ]);
  }
  const extra = cancelled.length > 0 ? `另外取消了 ${cancelled.length} 則排隊中的訊息。` : "";
  return `已停止。${extra}`;
}

- `levels`：兩個工具都是「查閱」（只查資料、不影響外界）。
- 沒有 `SERPAPI_KEY`：不註冊工具、`prompt` 為空，印「外掛「地圖」：.env 沒有 SERPAPI_KEY，先不提供查地圖。」
- 快取：`data/maps-cache/<雜湊>.json`，7 天；`usage.json` 記當天真查詢次數，超過 30 次丟錯誤「今天查地圖已經 30 次，到每日上限了，明天再查。沒查到的車程、營業時間不要自己估，照實告訴使用者沒查到。」，這一輪回覆最後 📍 那行下面加「⚠️ 今天查地圖已到每日上限（30 次），沒查到的地點、車程是秘書估的，請自己確認。」（V1.1；真機曾在額度用完後，行程裡的車程全是模型估的，回覆卻沒提醒）。快取不存 `search_metadata`。
- Google 地圖沒有營業時間 → 回傳「Google 地圖沒有列出營業時間（不要自己推測，請使用者自行確認）」。
- **班次（V1.1）**：SerpApi 的路線結果裡，大眾運輸有真實班次（例：「Tze-Chiang Limited Express 418 往臺東」花蓮 上午11:51 → 台東 下午1:52，含停靠站）。`transitPlan()` 整理成出發抵達、轉乘次數、每一段；`trainTitle()` 把區間車、自強號、莒光號、高鐵、太魯閣號、普悠瑪號翻成中文；`clock()` 把「下午1:11」轉成「13:11」。`depart_at` 送 SerpApi 的 `time=depart_at:<秒>`：**實測 SerpApi 把時間戳當成當地的鐘面時間**，送真正的台北時間 08:00（UTC 00:00）會查到凌晨的班次，所以用 `Date.UTC(年, 月, 日, 時, 分)` 編碼（`departAt()`）。起因：真機排台中出發的東台灣行程，秘書沒查班次就寫「09:00–11:30 花蓮火車抵達」，但台中到花蓮實際要約 4 小時、轉 2 次車。

**為什麼要程式寫的查詢紀錄**（紀律第 3 條）：真金鑰實測，Google 地圖沒有列出營業時間的地點，模型會自己補上時間；收緊規則後大多會寫「請自行確認」，但有一次模型根本沒查地圖，直接寫「歌劇院星期一 11:30–21:00（資料來自 Google 地圖）」。提示詞無法保證，所以由程式留下模型改不了的紀錄：

- 每次查詢（含快取）在黑視窗印「查地圖：…（新查詢／快取）」。
- 記下這一輪（`agent_start` 重置）查過的地點、路線；營業時間縮成一句（`compactHours()`：「星期一 休息；其他天 09:00–17:00」）。
- 秘書寫入 `文件/…行程….md` 後（`tool_result`），在檔尾附上或更新「## 地圖查詢紀錄（系統自動附上）」。
- 每次查詢後用 `bridge.setReplyNote("地圖", "📍 本次實際查了 N 個地點、M 段路線（Google 地圖，查詢於 …）")` 更新回覆最後一行。沒查、也沒寫行程檔就沒有。
- **行程比對（V1.1）**：寫入行程檔後，`auditTrip()` 數行程裡寫了幾處營業時間、幾處車程（秘書自己標「未查」「估計」「沒有列出」「自行確認」的不算），跟這一輪實際查的地點數、路線數比；寫的比查的多，就在回覆與檔尾紀錄加「⚠️ 行程寫了 N 處營業時間，這一輪只查了 M 個地點；其他的是秘書寫的，沒有查證，請自行確認。」（車程同理）。這一輪沒查地圖，檔尾紀錄寫「這一輪沒有查 Google 地圖。」。**有沒查證的時間時，還會把寫檔的工具結果退回給秘書**：附上「【系統檢查】…請用 maps_place 查每一站、maps_route 查車程，再用 edit 改進行程檔；查不到的改寫成未查、估計」，每一輪只退回一次（避免來回不停），撞到每日上限就不退回。秘書補查改檔後會重新比對，⚠️ 就消失。起因：加了 ⚠️ 之後真機排越南、泰國、印尼 9 天行程，秘書有查地圖的工具，卻選了「我沒辦法上網查即時資料」的提醒，一次都沒查。外掛的 `prompt` 也加上「排行程時你有這兩個工具（國內外都查得到），每一站都要查」，以及「要先照 trip-plan 技能問清楚再查」「搭火車、高鐵、客運的段落用 `depart_at` 查班次」。起因：真機排 9 天北歐行程只查了 2 個地點、0 段路線，其餘 16 處營業時間、7 處車程都是模型寫的；技能規則加強後，排大阪京都行程時一次都沒查，仍寫了 10 處營業時間。

/** 「上午8:04」「下午1:11」→「08:04」「13:11」；看不懂就照原樣。 */
function clock(text = "") {
  const match = /^(上午|下午|中午|凌晨|晚上)?\s*(\d{1,2}):(\d{2})$/.exec(text.trim());

  const hour = (Number(match[2]) % 12) + (["下午", "晚上", "中午"].includes(match[1]) ? 12 : 0);

}

/** 大眾運輸的一個選項：出發→抵達、要轉幾次車、每一段搭什麼。 */
function transitPlan(route) {
  const rides = (route.trips ?? []).filter((trip) => trip.travel_mode !== "Walking");
  return {
    出發抵達: `${clock(rides[0]?.start_stop?.time)} → ${clock(rides.at(-1)?.end_stop?.time)}`,
    時間: minutesText(route.duration, route.formatted_duration),
    轉乘: rides.length > 1 ? `${rides.length - 1} 次` : "不用轉車",
    分段: (route.trips ?? []).map((trip) => trip.travel_mode === "Walking"
      ? `步行 ${trip.formatted_duration ?? ""}`.trim()
      : `${trainTitle(trip.title)}：${trip.start_stop?.name} ${clock(trip.start_stop?.time)} → ${trip.end_stop?.name} ${clock(trip.end_stop?.time)}`),
  };
}

/** 「2026-10-24 08:00」（當地時間）→ SerpApi 的 depart_at。SerpApi 把時間戳當成當地的鐘面時間（實測），所以用 UTC 編碼。 */
function departAt(text) {
  const match = /^(\d{4})-(\d{2})-(\d{2}) (\d{2}):(\d{2})$/.exec(text.trim());
  if (!match) throw new Error(`出發時間要寫成「2026-10-24 08:00」，收到「${text}」`);
  const [, y, mo, d, h, mi] = match.map(Number);
  return `depart_at:${Date.UTC(y, mo - 1, d, h, mi) / 1000}`;
}

class LimitError extends Error {} // 到每日上限（跟網路錯誤分開，回覆要特別提醒）

const today = () => new Date().toLocaleDateString("sv-SE"); // YYYY-MM-DD（本地時間）

/** 呼叫 SerpApi；有新鮮的快取就用快取。回傳 { data, queriedAt }。 */
async function serp(cacheDir, params) {
  mkdirSync(cacheDir, { recursive: true });
  const hash = createHash("sha1").update(JSON.stringify(params)).digest("hex").slice(0, 16);
  const cacheFile = join(cacheDir, `${hash}.json`);
  if (existsSync(cacheFile)) {
    const cached = JSON.parse(readFileSync(cacheFile, "utf8"));
    if (Date.now() - cached.savedAt < CACHE_DAYS * 86_400_000) return { data: cached.data, queriedAt: cached.queriedAt, cached: true };
  }
  const usageFile = join(cacheDir, "usage.json");
  const usage = existsSync(usageFile) ? JSON.parse(readFileSync(usageFile, "utf8")) : {};
  const count = usage.date === today() ? usage.count : 0;
  if (count >= DAILY_LIMIT) {
    throw new LimitError(`今天查地圖已經 ${DAILY_LIMIT} 次，到每日上限了，明天再查。沒查到的車程、營業時間不要自己估，照實告訴使用者沒查到。`);
  }
  writeFileSync(usageFile, JSON.stringify({ date: today(), count: count + 1 }));

  const url = `https://serpapi.com/search.json?${new URLSearchParams({ ...params, api_key: KEY })}`;
  let data;
  try {
    const response = await fetch(url, { signal: AbortSignal.timeout(20_000) });
    data = await response.json();
  } catch (error) {
    throw new Error(`連不上地圖服務（${error.name === "TimeoutError" ? "逾時" : "網路錯誤"}），請稍後再試。`);
  }
  if (data.error) throw new Error(`地圖查不到：${data.error}`);
  delete data.search_metadata; // 不存查詢紀錄的連結
  const queriedAt = today();
  writeFileSync(cacheFile, JSON.stringify({ savedAt: Date.now(), queriedAt, data }));
  return { data, queriedAt, cached: false };
}

/** 一週的營業時間：{ 星期日: "09:00–17:00", … }；Google 地圖沒有就回傳說明文字。 */
function weeklyHours(place) {
  if (place.operating_hours && typeof place.operating_hours === "object") return place.operating_hours;
  if (Array.isArray(place.hours)) return Object.assign({}, ...place.hours);
  return "Google 地圖沒有列出營業時間（不要自己推測，請使用者自行確認）";
}

function summarizePlace(place) {
  return {
    名稱: place.title,
    地址: place.address,
    類型: [place.type ?? place.types ?? []].flat().join("、") || undefined,
    評分: place.rating ? `${place.rating}（${place.reviews ?? "?"} 則評論）` : undefined,
    營業時間: weeklyHours(place),
    票價: place.admission?.[0]?.options?.map((option) => `${option.title ?? ""} ${option.price ?? ""}`.trim()).join("、") || undefined,
    電話: place.phone,
  };
}

const text = (value) => ({ content: [{ type: "text", text: JSON.stringify(value, null, 1) }], details: {} });

/** 一週營業時間縮成一句：「每天 09:00–17:00」或「星期一 休息；其他天 09:00–17:00」。 */
function compactHours(hours) {
  if (typeof hours !== "object") return "Google 地圖沒有列出營業時間";
  const groups = new Map();
  for (const [day, time] of Object.entries(hours)) groups.set(time, [...(groups.get(time) ?? []), day]);
  const [common] = [...groups.entries()].sort((a, b) => b[1].length - a[1].length)[0];
  if (groups.size === 1) return `每天 ${common}`;
  const others = [...groups.entries()].filter(([time]) => time !== common).map(([time, days]) => `${days.join("、")} ${time}`);
  return `${others.join("；")}；其他天 ${common}`;
}

const TRIP_FILE = /^文件\/[^/]*行程[^/]*\.md$/;
const LOG_HEADING = "## 地圖查詢紀錄（系統自動附上）";

/** 比對行程檔：寫了幾處營業時間、車程，這一輪實際查了幾個地點、幾段路線；寫的比查的多，多出來的就是沒查證的。 */
function auditTrip(body, counts) {
  const lines = body.split("\n");
  const marked = /未查|估計|沒有列出|自行確認/; // 秘書自己標了沒查證的，不算
  const hours = lines.filter((line) => line.includes("營業時間") && !marked.test(line)).length;
  const drives = lines.filter((line) => /車程|🚗|🚆/.test(line) && /\d+\s*(分|小時)/.test(line) && !marked.test(line)).length;
  const warnings = [];
  if (hours > counts.地點) warnings.push(`⚠️ 行程寫了 ${hours} 處營業時間，這一輪只查了 ${counts.地點} 個地點；其他的是秘書寫的，沒有查證，請自行確認。`);
  if (drives > counts.路線) warnings.push(`⚠️ 行程寫了 ${drives} 處車程，這一輪只查了 ${counts.路線} 段路線；其他的是秘書估的，請自行確認。`);
  return warnings;
}

### 3.6 工序六：升級 trip-plan 技能

跟 L2-4 版相比：第 3 步改成「有 `maps_place`、`maps_route` 時每一站都要查」（當天沒營業就換掉；名稱對不上或有多個候選先問；查不到營業時間不可補；車程照查到的排）；每站多「營業時間」、站與站之間多「車程」；格式多「日期」；提醒分成「有查地圖時」「沒有查地圖工具時」兩種。

**V1.1 加強**（真機排跨國行程時，模型自己挑出發日、目的地寫「A 或 B」、編航班時刻、把冰島的教堂排進哥本哈根、沒查的營業時間寫得像查過）：確認事項加「節奏」，使用者問何時去要先給建議；跨國行程先定交通再排景點，目的地寫死一個，航班、船班、導覽團寫「待查」不編時刻；長途抵達日排輕鬆；看天氣的活動附備案；每站要在當天的城市，查地圖時核對地址；額度不夠時的優先順序；沒查的標「未查」「估計」；寫完自己檢查；提醒改成「只有地圖查詢紀錄列出的才是查到的」；回覆列出還要使用者確認的事。這些靠提示詞，Flash-Lite 不一定每次照做，所以營業時間、車程另外由程式比對（見 3.5）。

**再補三點**（真機繼續測試後）：①**先問再查再排**：原本「缺的一次問完；沒回答的用合理預設」，模型直接走後半句，連問都沒問；改成「這一則回覆只問問題，不查地圖、不排、不存檔」，附一段完整的提問範例（L2-6 的經驗：對 Flash-Lite 範例最有效），改完真機會先問 5 題、回答後才查。②**火車、高鐵、客運用 `maps_route` 加 `depart_at` 查參考班次**，查不到才寫待查。③排台中出發 2 大 2 小的東台灣行程時發現：寫了「七星潭適合親子踏浪」（浪大，不適合）、交通與入住也寫營業時間、搭火車卻都標 🚗、自己又寫一份查詢紀錄；技能加上親子長輩的安全提醒與行李提醒、營業時間只寫景點店家、交通符號依方式、不要自己寫查詢紀錄。

name: trip-plan
description: 依地點、天數、交通方式與喜好，規劃按時段排好的旅遊或出差行程表。使用者說「幫我排行程」「規劃一日遊」「出差去某地要怎麼安排」時使用。

# 行程規劃

## 步驟

1. **先問，再查、再排。** 要知道六件事：**地點、出發日期、天數、交通方式、風格或目的、同行人數與節奏（輕鬆或緊湊）**。
   - 使用者沒講到的，**這一則回覆只問問題，一次問完**：不查地圖、不排行程、不存檔。等他回答再開始。
   - 使用者回「都可以」「你決定」的項目，才用合理預設，並在行程開頭註明「（假設）」。
   - **使用者問「什麼時候去比較好」**：先回答建議的月份與理由（例如乾季雨季、颱風季、節慶、旺季人潮），請他選定日期，**不要自己挑日期直接排**。
   - 範例：使用者說「幫我規劃5天的越南旅遊」，只講了地點和天數，回覆應該是：

     > 好的！排之前想先確認幾件事：
     > 1. 預計哪一天出發？（還沒決定的話，我可以先建議適合的月份）
     > 2. 想去哪些城市？（例如胡志明市、河內、峴港，還是交給我建議）
     > 3. 交通方式：大眾運輸、包車，還是自己騎車？
     > 4. 旅遊風格：美食、文化古蹟、海島度假，還是都要一點？
     > 5. 幾個人去？喜歡輕鬆慢遊，還是行程排滿？
     > 回答後我就開始查地圖、排行程。
2. **跨國、跨城市的行程，先把每一段的交通定下來，再排景點**：
   - 每個地區寫**一個確定的城市或小鎮**，不可以寫「A 或 B」。不確定去哪裡，先列出選項和差別問使用者。
   - **火車、高鐵、客運**：有 maps_route 時，填 `depart_at`（出發日期與時間）查班次，寫「🚆 參考班次：自強號 418，花蓮 11:51 → 台東 13:52（查詢於 YYYY-MM-DD，出發前請到台鐵確認）」；要轉車的每一段都寫。沒查到的寫「🚆 班次待查（約 N 小時，是否要轉車）」。
   - **飛機、渡輪、船班、導覽團我查不到班表**：寫「✈️ 航班待查（航程約 N 小時）」，不要自己編起飛、抵達時間；抵達後的活動寫「抵達後」，不要寫死時刻。
   - 長途飛行或夜間航班抵達的那一天，只排入住、吃飯和附近散步。
   - 看天氣才能成行的活動（出海跳島、浮潛、登山、看日出、極光等），寫「看天氣」並附一個備案。
   - **有小孩或長輩同行**：海邊、溪邊、步道等有危險的地方加一句安全提醒，不要寫「踏浪」「戲水」；要換旅館的日子，提醒行李怎麼處理（例如寄放車站）。
3. 先選好景點：同一時段的景點盡量在同一區，減少移動時間。**每一站都要在當天所在的城市**。
4. **有 maps_place、maps_route 工具時要查地圖**：
   - 用 maps_place 查營業時間，**核對查到的地址在當天的城市**；不在，就是查錯或排錯，換掉。**查到的名稱跟使用者說的對不上、或有好幾個候選時，先列出來問使用者是哪一個**，不要自己挑。**出發那天沒營業、或排的時段不在營業時間內，就換掉或調整時段**，並在回覆說明換了什麼。
   - 站與站之間用 maps_route 查使用者那種交通方式要多久，用查到的時間排時段。
   - 同一個地點、同一段路線只查一次。**每天最多查 30 次**，查不完時優先查：①有營業時間的地方（博物館、教堂、樂園、餐廳）②自駕路段。
   - **沒查的一律標出來**：營業時間寫「未查，請自行確認」（戶外景點也一樣，不要寫「全天開放」）；車程寫「約 N 分（估計，未查）」。**不可以用記憶補上營業時間，也不要加「通常」「大約」之類的推測**（讀的人會以為是查到的；那天剛好公休就白跑一趟）。
   - 存檔後程式會比對：行程寫的營業時間、車程比實際查的多，回覆和檔案最後會自動加 ⚠️，使用者一看就知道哪些沒查證。
5. 每一站寫：時段、名稱、特色、**營業時間**、交通或停車；站與站之間寫交通時間，符號依交通方式：🚗 開車、🛵 機車、🚆 火車或高鐵、🚌 公車或客運、🚶 步行。午餐、晚餐要排進去。
   - 「營業時間」只寫景點和店家；搭車、入住飯店這類不用寫。
   - **不要自己寫「地圖查詢紀錄」**，程式會附在檔案最後。
6. 寫完自己檢查一遍：每一站在不在當天的城市？航班、船班有沒有編時刻？火車班次是查到的嗎？沒查的有沒有標「未查」「估計」？有小孩或長輩時，危險的地方有沒有提醒？
7. 照下方格式產出，存到 `文件/YYYY-MM-DD_行程_地點.md`（日期用出發日；不知道就用今天）。
8. 回覆使用者：行程摘要 + 存檔路徑 + 下方「提醒」（有換掉景點的話一起說明）＋**還沒確定、要使用者自己查或決定的事**（例如航班、船班、要去哪個小鎮）。
