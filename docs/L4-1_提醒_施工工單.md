## 1. 作業紀律

前面各課的紀律全部繼續有效。本課另外加上：

| # | 紀律 | 違反會怎樣 |
|---|---|---|
| 1 | **時間到的提醒由程式直接送出，不可以呼叫模型。** `tick` 讀 `提醒.json`，到時間就 `bridge.notify()`。 | 沒人看著時呼叫模型＝會用工具、會做事、會花錢，而且內容會被改寫 |

### 3.4 工序四：更新 `bridge.mjs`

**為什麼**：到目前為止，秘書只有在委託人傳訊息時才會動。這一課給主程式一個「心跳」：每 30 秒叫一次每個外掛的 `tick()`，外掛自己決定要不要做事。提醒外掛就是在 `tick` 裡比對時間、到了就送訊息。

**為什麼比對牆上時鐘，不用 `setTimeout`**（紀律第 4 條）：`setTimeout` 排到明天早上八點，電腦睡眠時計時器不會走，醒來後那個提醒就永遠不會發。每 30 秒去看一次「現在幾點、有沒有已經過了還沒送的」，睡醒後第一次 `tick` 就會補發。

用以下參考件**整份取代** `bridge.mjs`。跟 `L3-7-V1.4` 相比的改動：

| 位置 | 改動 |
|---|---|
| `import` | 加 `appendFileSync` |
| 新增 事件紀錄 區段 | `EVENT_FILE = data/事件紀錄.jsonl`；`logEvent(kind, detail)` 一行一個 JSON，寫不進去只印錯誤不中斷。只記種類與摘要，**不記對話內容**；在 `data/` 裡不進 Git，秘書也讀不到 |
| `buildSystemPrompt()` | 依工具清單有沒有 `add_reminder` 切換那句話：有 → 「除了提醒…你不會到時間自己去做事」；沒有 → 沿用 L2-3 的「你不會主動發訊息…」。**拿掉提醒外掛就要恢復原本那句** |
| `onAgentEvent()` | `agent_error` 記「模型出錯」、`auto_retry_end` 記「自動重試」 |
| `pluginApi` | 加 `journalTodos()`（最新一份日誌的「還沒完成的事」，給待辦追蹤用）、`notify(text)`（主動傳訊息給主人，送不出去走既有的 outbox 補送） |
| `loadPlugins()` | 載入結果多帶一個 `tick: module.tick` |
| `permissionExtension` | 擋下時 `logEvent("擋下", { 工具, 原因, 指令 })`（指令只留前 100 字） |
| `stop()` | `logEvent("/stop", { 正在做事, 取消排隊 })` |
| 白名單後面 | `OWNER = [...ALLOWED][0]`：主動訊息只發給主人 |
| 啟動訊息 | 多一行「主動訊息發給：【ID】（TELEGRAM_ALLOWED_USERS 的第一個）」 |
| 檔案最後 | `runTicks()`：跑每個外掛的 `tick(pluginApi)`，上一輪沒做完就跳過這輪，單一外掛出錯只印錯誤。**先呼叫一次**（補發關機時錯過的），再 `setInterval(runTicks, 30000)` |

/**
 * 使用者說「這週六」「本週五」，而那天這週已經過了（例如星期日說「這週六」）→ 回傳一行給模型的提示。
 * 模型常把它當成下週，而且對話裡有前例時，system prompt 的規則壓不過，所以由程式比對、附在這則訊息後面。
 */
function pastWeekdayHint(text, now = new Date()) {
  const week = "日一二三四五六";
  const today = (now.getDay() + 6) % 7; // 星期一 = 0 … 星期日 = 6
  const label = (date) => `${date.getMonth() + 1}/${date.getDate()}（${week[date.getDay()]}）`;
  const hints = new Set();
  for (const match of text.matchAll(/(這|本)(週|周|星期|禮拜)([一二三四五六日天])/g)) {
    const target = (week.indexOf(match[3] === "天" ? "日" : match[3]) + 6) % 7;
    if (target >= today) continue;
    const date = new Date(now.getFullYear(), now.getMonth(), now.getDate() - (today - target));
    const next = new Date(date.getFullYear(), date.getMonth(), date.getDate() + 7);
    hints.add(`「${match[0]}」是 ${label(date)}，已經過了。請用你自己的話問使用者是不是指下週的 ${label(next)}；他確認之前不要用工具去做。`);
  }
  return hints.size > 0 ? `\n【系統提醒：只給你看，不要照抄給使用者】${[...hints].join("")}` : "";
}

let replyBuffer = "";
let lastAgentError = "";
function onAgentEvent(event) {
  if (event.type === "message_update" && event.assistantMessageEvent.type === "text_delta") {
    replyBuffer += event.assistantMessageEvent.delta;
  }
  if (event.type === "message_end" && event.message.role === "assistant" && event.message.errorMessage) {
    lastAgentError = event.message.errorMessage;
    logEvent("模型出錯", { 訊息: event.message.errorMessage.slice(0, 200) });
  }
  if (event.type === "auto_retry_end" && !event.success && event.finalError) {
    lastAgentError = event.finalError;
  }
  if (event.type === "auto_retry_end") logEvent("自動重試", { 成功: event.success, 第幾次: event.attempt });
}

/** 讀取 外掛/ 底下每個資料夾的 index.mjs；載入失敗的外掛略過，不影響其他功能。 */
async function loadPlugins() {
  if (!existsSync(PLUGIN_DIR)) return [];
  const loaded = [];
  for (const entry of readdirSync(PLUGIN_DIR, { withFileTypes: true })) {
    const file = join(PLUGIN_DIR, entry.name, "index.mjs");
    if (!entry.isDirectory() || !existsSync(file)) continue;
    try {
      const module = await import(pathToFileURL(file).href);
      if (typeof module.default !== "function") throw new Error("index.mjs 要用 export default 匯出一個函式");
      loaded.push({
        name: entry.name,
        prompt: module.prompt ?? "",
        levels: module.levels ?? {}, // 外掛宣告自己的工具屬於哪一級；沒宣告的算「執行」
        onFile: module.onFile, // 使用者傳照片、檔案時由外掛處理（讀檔案）；沒有外掛處理就照舊回「看不懂」
        describe: module.describe ?? {}, // 執行級工具的確認說明：{ 工具名稱: async (input) => 白話文字 }
        shellWarnings: module.shellWarnings ?? [], // 指令碰到這個外掛的東西時，按鈕上方加的警示：[{ pattern, text }]
        tick: module.tick, // 每 30 秒由主程式呼叫一次（例如到時間的提醒）；不排隊、不經模型
        extension: { name: entry.name, factory: (pi) => module.default(pi, pluginApi) },
      });
    } catch (error) {
      failedPlugins.push(entry.name);
      console.error(`外掛「${entry.name}」載入失敗，先略過：`, errorText(error));
    }
  }
  return loaded;
}
const plugins = await loadPlugins();

// ── 權限分級：查閱直接做、擬稿限定位置、執行先問使用者（規則在 權限.json）──────
const PERMISSION_FILE = join(ROOT, "權限.json");
const DEFAULT_PERMISSIONS = {
  可以寫的地方: ["文件/", ".pi/skills/"],
  只能修改的檔案: ["MEMORY.md"],
  不能讀的地方: [".env", "data/"],
  等我確認幾分鐘: 5,
};
const LEVELS = { read: "查閱", grep: "查閱", find: "查閱", ls: "查閱", write: "擬稿", edit: "擬稿" };

/** 讀 權限.json；沒有檔案或格式錯誤就用預設值。 */
function loadPermissions() {
  if (!existsSync(PERMISSION_FILE)) return DEFAULT_PERMISSIONS;
  try {
    const custom = JSON.parse(readFileSync(PERMISSION_FILE, "utf8"));
    const merged = { ...DEFAULT_PERMISSIONS, ...custom };
    const lists = ["可以寫的地方", "只能修改的檔案", "不能讀的地方"];
    if (!lists.every((key) => Array.isArray(merged[key]) && merged[key].every((item) => typeof item === "string")) ||
      !(merged.等我確認幾分鐘 > 0)) {
      throw new Error("欄位格式不對");
    }
    return merged;
  } catch (error) {
    console.error("權限.json 讀不懂，先用預設權限：", errorText(error));
    return DEFAULT_PERMISSIONS;
  }
}
const permissions = loadPermissions();

/** 工具屬於哪一級：內建工具固定；外掛工具看外掛宣告，沒宣告的算「執行」。 */
function levelOf(toolName) {
  return LEVELS[toolName] ?? plugins.map((plugin) => plugin.levels[toolName]).find(Boolean) ?? "執行";
}

/** 路徑換成相對於專案資料夾、用 / 分隔；在專案外面的回傳 null。 */
function projectPath(path) {
  const rel = relative(ROOT, resolve(ROOT, path ?? ".")).split(sep).join("/");
  return rel.startsWith("../") || rel === ".." || /^[a-zA-Z]:/.test(rel) ? null : rel;
}
const inside = (rel, places) => places.some((place) => place.endsWith("/") ? `${rel}/`.startsWith(place) : rel === place);

/**
 * 執行級指令的警示：提到秘書平常不能改的檔案、或是刪除類指令，就在按鈕上方加一行提醒。
 * 直接比對指令文字，不靠模型判斷；換個寫法可能比對不到，所以只是輔助，決定權仍在使用者。
 */
function warningsFor(command) {
  const writable = permissions.可以寫的地方.map((place) => place.split("/")[0]);
  const escape = (text) => text.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  const protectedNames = readdirSync(ROOT)
    .filter((name) => name !== "node_modules" && !writable.includes(name))
    .filter((name) => new RegExp(`(^|[\\\\/"'\\s])${escape(name)}($|[\\\\/"'\\s;,)])`, "i").test(command));
  const warnings = [];
  if (protectedNames.length > 0) warnings.push(`⚠️ 會碰到秘書平常不能改的檔案：${protectedNames.join("、")}`);
  if (/(^|[\s;|&(])(Remove-Item|rm|rmdir|rd|del|erase|ri)(\s|$)/i.test(command)) warnings.push("⚠️ 這個指令會刪除東西");
  for (const warning of plugins.flatMap((plugin) => plugin.shellWarnings)) {
    if (warning.pattern.test(command)) warnings.push(warning.text);
  }
  return warnings;
}

/** 檢查一次工具呼叫；回傳擋下的原因（寫成秘書可以直接轉告使用者的話），可以執行就回傳 undefined。 */
async function checkPermission(event) {
  const level = levelOf(event.toolName);
  if (level === "查閱") {
    const rel = projectPath(event.input.path);
    if (rel === null) return `我只能看專案資料夾裡的東西，沒有權限看 ${event.input.path}。`;
    if (inside(rel, permissions.不能讀的地方)) return `我沒有權限讀 ${rel}。`;
    return undefined;
  }
  if (level === "擬稿") {
    // 外掛的擬稿工具沒有寫檔路徑（例如 Gmail 草稿）：只做不影響外界的事，由外掛自己把關；有路徑的（例如 Office）照樣限定位置
    if (event.toolName !== "write" && event.toolName !== "edit" && event.input.path === undefined) return undefined;
    const rel = projectPath(event.input.path);
    if (rel !== null && inside(rel, permissions.可以寫的地方)) return undefined;
    if (rel !== null && event.toolName === "edit" && inside(rel, permissions.只能修改的檔案)) return undefined;
    return `我沒有權限${event.toolName === "write" ? "寫入" : "修改"} ${rel ?? event.input.path}：我只能寫 ${permissions.可以寫的地方.join("、")}，` +
      `${permissions.只能修改的檔案.join("、")} 只能修改、不能整份覆寫。`;
  }
  // 外掛可以提供白話的確認說明；說明做不出來（例如參數有錯、只能看的行事曆），就直接擋下，不問使用者
  const describe = plugins.map((plugin) => plugin.describe[event.toolName]).find(Boolean);
  let command = event.toolName === SHELL_TOOL ? event.input.command : JSON.stringify(event.input);
  if (event.toolName !== SHELL_TOOL && describe) {
    try {
      command = await describe(event.input);
    } catch (error) {
      return `我沒辦法做：${errorText(error)}`;
    }
  }
  const warnings = event.toolName === SHELL_TOOL ? warningsFor(command) : [];
  const answer = await askUser([`秘書要執行：${event.toolName}`, ...warnings, command.slice(0, 3000)].join("\n\n"));
  if (answer === "run") return undefined;
  return {
    skip: "你按了「不要」，所以我沒有執行。",
    timeout: `你 ${permissions.等我確認幾分鐘} 分鐘內沒有按按鈕，所以我沒有執行。`,
    stop: "你用 /stop 叫停了，所以我沒有執行。",
    none: "現在沒有人在跟我說話，所以我沒有執行。",
    failed: "確認訊息傳不出去，所以我沒有執行。",
  }[answer];
}

let pendingAsk = null; // 等使用者按按鈕的那一題 { resolve, chatId, messageId, text }
const ANSWER_NOTES = { run: "✅ 已同意執行", skip: "❌ 你按了不要", timeout: "⌛ 太久沒回應，沒有執行", stop: "⏹ 已用 /stop 取消" };

/** 傳一則附「執行／不要」按鈕的訊息，等使用者按；回傳 run、skip、timeout、stop、none 或 failed。 */
async function askUser(text) {
  if (!running) return "none";
  const chatId = running.chatId;
  let sent;
  try {
    sent = await tg("sendMessage", {
      chat_id: chatId,
      text,
      link_preview_options: { is_disabled: true },
      reply_markup: { inline_keyboard: [[{ text: "✅ 執行", callback_data: "run" }, { text: "❌ 不要", callback_data: "skip" }]] },
    });
  } catch (error) {
    console.error("確認訊息傳不出去：", errorText(error));
    return "failed";
  }
  const answer = await new Promise((resolve) => {
    const timer = setTimeout(() => resolve("timeout"), permissions.等我確認幾分鐘 * 60_000);
    pendingAsk = { resolve: (value) => { clearTimeout(timer); resolve(value); }, chatId, messageId: sent.result.message_id, text };
  });
  const { messageId } = pendingAsk;
  pendingAsk = null;
  // 把按鈕換成結果，舊按鈕就不能再按
  await tg("editMessageText", { chat_id: chatId, message_id: messageId, text: `${text}\n\n${ANSWER_NOTES[answer]}` })
    .catch((error) => console.error("更新按鈕訊息失敗：", errorText(error)));
  return answer;
}

/** 對話存檔檔名是「時間戳_工作日-序號.jsonl」，依日期、序號排序。 */
function sessionFiles() {
  return readdirSync(SESSION_DIR)
    .map((name) => {
      const match = name.match(/_(\d{4}-\d{2}-\d{2})-(\d+)\.jsonl$/);
      return match && { path: join(SESSION_DIR, name), day: match[1], seq: Number(match[2]) };
    })
    .filter(Boolean)
    .sort((a, b) => a.day.localeCompare(b.day) || a.seq - b.seq);
}

function isJournaled(manager) {
  return manager.getEntries().some((entry) => entry.type === "custom" && entry.customType === JOURNAL_DONE);
}

function hasConversation(manager) {
  return manager.getEntries().some((entry) => entry.type === "message" && entry.message.role === "user");
}

function textFromLatestAssistantMessage(session) {
  const message = [...session.messages].reverse().find((item) => item.role === "assistant");
  if (!message || !Array.isArray(message.content)) return "";
  return message.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("");
}

/** 把一本對話整理成 日誌/YYYY-MM-DD.md；同一天已有日誌就合併。由程式寫檔，不靠模型用工具。 */
async function writeJournal(target) {
  const file = `${target.day}.md`;
  const path = join(JOURNAL_DIR, file);
  const earlier = existsSync(path) ? readFileSync(path, "utf8").trim() : "";
  // 同一天稍早的日誌已經延續過待辦，只有這一天第一次整理時才需要帶上一份日誌的待辦。
  const previousFile = earlier ? undefined : journalFiles().filter((name) => name < file).at(-1);
  const previous = previousFile && {
    day: previousFile.replace(/\.md$/, ""),
    text: readFileSync(join(JOURNAL_DIR, previousFile), "utf8"),
  };
  replyBuffer = "";
  lastAgentError = "";
  await promptSession(target.session, journalPrompt(target.day, earlier, previous));
  const text = textFromLatestAssistantMessage(target.session)
    .trim()
    .replace(/^```(?:markdown|md)?\s*\n/, "")
    .replace(/\n```\s*$/, "")
    .trim();
  if (!JOURNAL_HEADINGS.every((heading) => text.includes(heading))) {
    throw new Error(lastAgentError || "日誌格式不完整");
  }
  writeFileSync(path, `${text}\n`, "utf8");
  target.manager.appendCustomEntry(JOURNAL_DONE, { file: `日誌/${file}` });
  return `日誌/${file}`;
}

/** 把還沒整理過的舊對話都寫成日誌（例如昨天關機前沒整理到的）。失敗的下次再試。 */
async function journalPendingSessions(exceptPath) {
  const written = [];
  for (const file of sessionFiles()) {
    if (stopRequested()) break; // 使用者傳了 /stop，剩下的留到下次再補
    if (file.path === exceptPath) continue;
    const manager = SessionManager.open(file.path, SESSION_DIR);
    if (isJournaled(manager) || !hasConversation(manager)) continue;
    const target = await openSession(manager, file.day);
    try {
      written.push(await writeJournal(target));
      console.log(`已整理工作日誌：日誌/${file.day}.md`);
    } catch (error) {
      console.error(`整理 ${file.day} 的工作日誌失敗，下次再試：`, errorText(error));
    } finally {
      target.session.dispose();
    }
  }
  return [...new Set(written)];
}

let current = null;
/** 上班：今天還有沒整理的那本就接著用，否則開新的一本；開之前先把舊的整理成日誌。 */
async function startWorkday() {
  const day = workday();
  const last = sessionFiles().filter((file) => file.day === day).at(-1);
  const lastManager = last && SessionManager.open(last.path, SESSION_DIR);
  const resume = lastManager && !isJournaled(lastManager) ? lastManager : null;
  const written = await journalPendingSessions(resume ? last.path : undefined);
  const manager = resume ?? SessionManager.create(ROOT, SESSION_DIR, { id: `${day}-${(last?.seq ?? 0) + 1}` });
  current = await openSession(manager, day);
  return written;
}

function errorText(error) {
  if (error instanceof Error) return error.message;
  if (typeof error === "string") return error;
  try {
    return JSON.stringify(error);
  } catch {
    return String(error);
  }
}

function readableAgentError(error) {
  const raw = [errorText(error), lastAgentError].filter(Boolean).join("\n").trim();
  if (/quota|rate.?limit|usage|429|too many|capacity|exceed|billing|resource_exhausted/i.test(raw)) {
    return "Gemini 免費額度或速率上限可能已到（Flash-Lite 免費層每分鐘、每天都有上限）。" +
      "系統已自動重試過幾次仍失敗，請稍等一分鐘再試；如果常常發生，建議到 Google AI Studio 開通計費解除限制。";
  }
  return raw
    ? `Agent 暫時無法回覆：${raw.slice(0, 1000)}`
    : "Agent 暫時無法回覆，請稍後再試。";
}

async function chat(text, file) {
  let notice = "";
  if (current.day !== workday()) {
    current.session.dispose();
    const written = await startWorkday();
    if (written.length > 0) notice = `（新的一天，已整理好工作日誌：${written.join("、")}）\n\n`;
  }
  // 使用者傳的照片、檔案：外掛下載、轉成文字，換掉 text；照片另外當附件給模型看
  let images;
  if (file) ({ text, images } = await file());
  replyBuffer = "";
  lastAgentError = "";
  if (stopRequested()) return notice;
  const leafBefore = current.manager.getLeafId();
  try {
    await promptSession(current.session, `【現在時間：${nowLabel()}】\n${text}${pastWeekdayHint(text)}`, images);
    // 只送最後一則回覆：模型在呼叫工具前寫的規劃文字不給使用者看。
    const reply = (textFromLatestAssistantMessage(current.session).trim() || replyBuffer.trim())
      .replace(/^【現在時間：[^】]*】\s*/, ""); // 模型偶爾會照抄訊息開頭的時間標記
    const notes = [...(running?.notes.values() ?? [])];
    return notice + (reply || readableAgentError()) + (notes.length > 0 ? `\n\n${notes.join("\n")}` : "");
  } catch (error) {
    return notice + readableAgentError(error);
  } finally {
    if (stopRequested()) await forgetStoppedTurn(leafBefore, text);
  }
}

/**
 * 被叫停的那一輪只留下做到一半的工具結果，秘書下一則常會接著回報、甚至說做完了。
 * 所以把對話退回這則訊息之前（原本的紀錄仍留在存檔的分支裡），改記一句「被叫停了」。
 */
async function forgetStoppedTurn(leafBefore, text) {
  try {
    const entries = current.manager.getEntries();
    const index = entries.findLastIndex((entry) => entry.parentId === leafBefore && entry.type === "message");
    const turn = index < 0 ? [] : entries.slice(index).map((entry) => entry.message).filter(Boolean);
    // 列出叫停前做過的動作（只列工具與參數，不放結果內容，免得秘書又去回報）
    const calls = new Map(turn.flatMap((message) => message.role === "assistant" && Array.isArray(message.content)
      ? message.content.filter((block) => block.type === "toolCall").map((block) => [block.id, block])
      : []));
    const done = turn.filter((message) => message.role === "toolResult").map((result) => {
      const output = JSON.stringify(result.content);
      const status = !result.isError ? "完成" : /aborted/i.test(output) ? "被中斷" : /沒有執行|沒有權限/.test(output) ? "沒有執行" : "失敗";
      const args = JSON.stringify(calls.get(result.toolCallId)?.arguments ?? {}).slice(0, 150);
      return `- ${result.toolName} ${args}：${status}`;
    });
    if (index >= 0) await current.session.navigateTree(entries[index].id);
    await current.session.sendCustomMessage({ customType: STOPPED, content: stoppedNote(text, done), display: false });
  } catch (error) {
    console.error("整理被叫停的對話失敗：", errorText(error));
  }
}

/** /new：今天先到這，整理日誌並開新的一本。 */

  await startWorkday();

}

// ── 緊急停止：記下正在做的事與排隊中的訊息，/stop 不排隊、直接叫停 ──────
let running = null;        // 正在處理的訊息 { stopped, chatId, notes }
const waiting = new Set(); // 排隊中的訊息
let activeSession = null;  // 正在呼叫模型的 session（對話本身，或正在整理日誌的那一本）
const stopRequested = () => running?.stopped === true;

/** 呼叫模型，並記下是哪個 session，/stop 才知道要中止誰。 */
async function promptSession(session, text, images) {
  activeSession = session;
  try {
    await session.prompt(text, images ? { images } : undefined);
  } finally {
    activeSession = null;
  }
}

let queue = Promise.resolve();
/** 把一則訊息交給秘書，回傳完整文字回覆（被 /stop 叫停的回傳 null）；訊息排隊、一次一則。 */
function handleMessage(text, chatId, file) {
  const command = text.trim().match(/^\/(new|recap)(?:@\w+)?$/)?.[1];
  const task = { stopped: false, chatId, notes: new Map() }; // chatId：外掛傳檔案給誰；notes：外掛要附在回覆最後的話
  waiting.add(task);
  const run = queue.then(async () => {
    waiting.delete(task);
    if (task.stopped) return null;
    running = task;
    try {
      let reply;
      if (command === "new") reply = await newConversation();
      else if (command === "recap") reply = recap();
      else reply = await chat(text, file);
      return task.stopped ? null : reply;
    } finally {
      running = null;
    }
  });
  queue = run.catch(() => {});
  return run;
}

/** /stop：立刻叫停正在做的事，並取消排隊中的訊息。 */
async function stop() {
  const cancelled = [...waiting].filter((task) => !task.stopped);
  for (const task of cancelled) task.stopped = true;
  const wasRunning = running !== null;
  if (!wasRunning && cancelled.length === 0) return "目前沒有在做的事。";
  if (wasRunning) {
    running.stopped = true;
    pendingAsk?.resolve("stop"); // 正在等使用者按按鈕的，一起取消
    // 等秘書停下最多 5 秒；就算中止過程出錯，也一定要回覆使用者。
    await Promise.race([
      Promise.resolve(activeSession?.abort()).catch((error) => console.error("中止時發生錯誤：", errorText(error))),
      new Promise((resolve) => setTimeout(resolve, 5000)),
    ]);
  }
  logEvent("/stop", { 正在做事: wasRunning, 取消排隊: cancelled.length });
  const extra = cancelled.length > 0 ? `另外取消了 ${cancelled.length} 則排隊中的訊息。` : "";
  return `已停止。${extra}`;
}

async function tg(method, body) {
  const upload = body instanceof FormData; // 傳檔案用 multipart，其他用 JSON
  const response = await fetch(`${API}/${method}`, {
    method: "POST",
    ...(upload ? { body } : { headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) }),
    signal: AbortSignal.timeout(upload ? 120_000 : 45_000),
  });
  const raw = await response.text();
  let data;
  try {
    data = JSON.parse(raw);
  } catch {
    throw new Error(`Telegram 回應不是 JSON（HTTP ${response.status}）`);
  }
  if (!response.ok || !data.ok) {
    throw new Error(data.description || `Telegram API 錯誤（HTTP ${response.status}）`);
  }
  return data;
}

/**
 * 模型的回覆常用 Markdown（**粗體**、# 標題、- 清單、| 表格 |），Telegram 用純文字送出會滿是星號。
 * 這裡換成 Telegram 看得懂的樣子：文字去掉記號，粗體、斜體、程式碼、連結改用 entities 標位置
 * （不用 parse_mode：模型寫錯一個記號，整則就送不出去）。回傳 { text, entities }。
 */
function telegramFormat(markdown) {
  let out = "";
  const entities = [];
  const mark = (type, start, extra = {}) => {
    if (out.length > start) entities.push({ type, offset: start, length: out.length - start, ...extra });
  };
  /** 一行裡的粗體、斜體、刪除線、`程式碼`、[連結](網址)。 */
  const inline = (line) => {
    const pattern = /`([^`\n]+)`|\*\*(.+?)\*\*|__(.+?)__|~~(.+?)~~|\[([^\]\n]+)\]\((https?:\/\/[^)\s]+)\)|(?<![\w*])\*(?!\s)([^*\n]+?)(?<!\s)\*(?![\w*])/g;
    let last = 0;
    for (const m of line.matchAll(pattern)) {
      out += line.slice(last, m.index);
      const start = out.length;
      if (m[1] !== undefined) { out += m[1]; mark("code", start); }
      else if (m[2] !== undefined || m[3] !== undefined) { inline(m[2] ?? m[3]); mark("bold", start); }
      else if (m[4] !== undefined) { inline(m[4]); mark("strikethrough", start); }
      else if (m[5] !== undefined) { out += m[5]; mark("text_link", start, { url: m[6] }); }
      else { inline(m[7]); mark("italic", start); }
      last = m.index + m[0].length;
    }
    out += line.slice(last);
  };
  const lines = markdown.split("\n");
  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];
    if (i > 0) out += "\n";
    // 程式碼區塊：原樣、等寬
    if (/^\s*```/.test(line)) {
      const start = out.length;
      const body = [];
      for (i++; i < lines.length && !/^\s*```/.test(lines[i]); i++) body.push(lines[i]);
      out += body.join("\n");
      mark("pre", start);
      continue;
    }
    const heading = line.match(/^ {0,3}#{1,6} +(.*?) *#* *$/);
    if (heading) {
      const start = out.length;
      inline(heading[1]);
      mark("bold", start);
      continue;
    }
    if (/^\s*([-*_])(\s*\1){2,}\s*$/.test(line)) { out += "──────────"; continue; } // 分隔線
    if (/^\s*\|.*\|\s*$/.test(line)) {
      if (/^\s*\|?\s*:?-{2,}:?\s*(\|\s*:?-{2,}:?\s*)*\|?\s*$/.test(line)) { // 表格的 |---| 那一列：拿掉，上一列（表頭）改粗體
        out = out.slice(0, -1);
        mark("bold", out.lastIndexOf("\n") + 1);
        continue;
      }
      inline(line.trim().replace(/^\|\s*|\s*\|$/g, "").split(/\s*(?<!\\)\|\s*/).join("｜").replace(/\\\|/g, "|"));
      continue;
    }
    // 只認半形空白：程式寫的訊息（例如 /status）用全形空白縮排，不要動
    const bullet = line.match(/^( *)[-*+] +(.*)$/);
    if (bullet) {
      out += `${"　".repeat(Math.floor(bullet[1].length / 2))}• `;
      inline(bullet[2]);
      continue;
    }
    const quote = line.match(/^ *> ?(.*)$/);
    inline(quote ? `｜${quote[1]}` : line);
  }
  return { text: out, entities };
}

/** 要送出的內容：Markdown 換成純文字＋entities；不顯示連結預覽卡（回覆常附網址，預覽卡很佔版面）。 */
function messageBody(chatId, text) {
  const { text: plain, entities } = telegramFormat(text.slice(0, 4000));
  return { chat_id: chatId, text: plain, ...(entities.length > 0 ? { entities } : {}), link_preview_options: { is_disabled: true } };
}

### 3.5 工序五：新增 `台灣假日.mjs`

**為什麼**：待辦追蹤在週末與國定假日不該吵委託人。台灣的補假、調整放假每年不同，寫死在程式裡會過期，所以從 Google 的台灣假日行事曆抓，一天最多抓一次，存在 `data/台灣假日.json`。抓不到就用上次存的，一小時後再試——**沒有假日資料時一律當成上班日**，寧可多提醒一次，也不要該提醒時不提醒。

只取 `DESCRIPTION:國定假日` 的那些；情人節、冬至這類「假日節慶」照常上班，不算。

### 3.6 工序六：新增 `外掛/提醒/index.mjs`

**為什麼**：提醒是這門課第一個「不等人開口」的功能，但它刻意做得最小——時間到只送出一段固定文字，不呼叫模型、不碰任何工具。這樣最穩，也讓委託人先習慣「秘書會主動出聲」這件事，之後 L4-2 才讓模型在沒人看著時做事。

這個外掛有兩組工具：

**提醒**（三個）

| 工具 | 權限級 | 說明 |
|---|---|---|
| `add_reminder` | 擬稿 | 一次性填 `at`（`YYYY-MM-DDTHH:mm`），循環填 `repeat`（每天／每個工作日／每週X／每月N日 `HH:mm`），兩者只能填一個 |
| `list_reminders` | 查閱 | 依時間列出；每行最後的〔代號〕只給 `cancel_reminder` 用 |
| `cancel_reminder` | 擬稿 | 傳〔代號〕 |

**待辦追蹤**（四個，全是擬稿級）

| 工具 | 說明 |
|---|---|
| `track_todos` | 打開／關掉，順便改時間設定 |
| `list_todos` | 今天還有哪些沒做 |
| `update_todo` | 回報某一件已完成／擱置 |
| `add_todo` | 臨時加一件到今天的清單 |

全部是擬稿級、不跳按鈕：這些工具只會在時間到時傳一段**委託人自己說過的話**給**他本人**，不影響外界。

要注意的幾個實作點：

| 項目 | 做法 | 為什麼 |
|---|---|---|
| 送出順序 | 先寫 `提醒.json`，再 `await bridge.notify()` | 送出途中當掉不會重複提醒（紀律第 2 條） |
| 補發 | 晚超過 2 分鐘才送出的，訊息後面加「（原定 …，那時秘書沒有在執行，現在補發）」 | 委託人要看得出來是補的 |
| 循環的補發 | 送出後排下一次，所以只補最近那一次 | 關機三天不會一次倒三天份 |
| 代號 | `提醒.json` 存 `下一個編號`，只進不退 | 紀律第 6 條 |
| 壞檔案 | `提醒.json` 格式壞掉時：略過壞掉的那幾筆、照常運作，存檔時原樣留著，並在工具被呼叫時丟出錯誤請委託人修 | 不要把委託人的資料覆蓋掉 |
| 待辦清單來源 | 每天第一次提醒前，用 `bridge.journalTodos()` 從最新日誌的「還沒完成的事」重列；狀態只記在 `待辦.json` | 不改工作日誌（L2-3 的做法不變） |
| 放假 | `只在工作日`（預設 `true`）：週末與國定假日不提醒、不給收工清單 | 放假不用被工作追著跑 |
| 安靜時段 | 讀 `排程.json` 的「安靜時段」；**`排程.json` 要 L4-2 才有**，沒有時用預設 `22:00-07:00`、`12:00-13:00` | 這一課不建 `排程.json`，不是缺東西 |
| 剛打開時 | 今天已經過掉的時間點標成做過，不補發 | 不然晚上打開會馬上收到「17:30 還沒完成」跟收工清單（實測踩過） |

```js
  throw new Error(`算不出「${repeat}」的下一次`);
}

const TODO_FILE = "待辦.json";

/** 今天放假嗎？放假回傳原因（「週末」或假日名稱），上班日回傳 null。 */
function restDay(root, day) {
  if (day.getDay() === 0 || day.getDay() === 6) return "週末";
  return holidayOn(root, day) ?? null;
}
const CLOCK = /^([01]\d|2[0-3]):([0-5]\d)$/;
const ymd = (date) => `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`;
const minutesOf = (text) => +text.slice(0, 2) * 60 + +text.slice(3);

function readTodos(root) {
  const path = join(root, TODO_FILE);
  if (!existsSync(path)) return { 追蹤: { ...TODO_DEFAULT }, 日期: null, 待辦: [] };
  try {
    const data = JSON.parse(readFileSync(path, "utf8"));
    const track = { ...TODO_DEFAULT, ...(data.追蹤 ?? {}) };
    if (!CLOCK.test(track.從幾點開始) || !CLOCK.test(track.收工時間) || !(track.每幾小時 > 0)) throw new Error("追蹤的時間設定不對");
    return { ...data, 追蹤: track, 待辦: Array.isArray(data.待辦) ? data.待辦.filter((item) => item && typeof item.內容 === "string") : [] };
  } catch (error) {
    console.error(`待辦.json 讀不懂，先不追蹤待辦（修好後不用重啟）：${error.message}`);
    return null;
  }
}

/** 今天的提醒時間點（從幾點開始、每幾小時，到收工時間為止，不含收工那一刻）。 */
function checkpoints(track) {
  const start = minutesOf(track.從幾點開始), end = minutesOf(track.收工時間);
  const list = [];
  for (let at = start; at < end && list.length < 24; at += track.每幾小時 * 60) list.push(at);
  return list;
}

/** 安靜時段（跟例行工作共用 排程.json 的設定）：落在裡面就延到時段結束。 */
function afterQuiet(root, minutes) {
  let ranges = ["22:00-07:00", "12:00-13:00"];
  try {
    const custom = JSON.parse(readFileSync(join(root, "排程.json"), "utf8")).安靜時段;
    if (Array.isArray(custom) && custom.length > 0) ranges = custom;
  } catch {}
  for (const range of ranges) {
    const match = String(range).match(/^(\d\d:\d\d)-(\d\d:\d\d)$/);
    if (!match) continue;
    const from = minutesOf(match[1]), to = minutesOf(match[2]);
    const inside = from <= to ? minutes >= from && minutes < to : minutes >= from || minutes < to;
    if (inside) return to;
  }
  return minutes;
}

      // 剛打開（或剛改時間）：今天已經過掉的時間點不補發，不然晚上打開會馬上收到「17:30 還沒完成」和收工清單
      const now = new Date();
      const rest = track.只在工作日 !== false && restDay(bridge.root, now);
      const minutes = now.getHours() * 60 + now.getMinutes();
      const points = checkpoints(track).map((at) => ({ at, real: afterQuiet(bridge.root, at) }));
      const passed = points.filter((point) => minutes >= point.real).at(-1);
      if (passed) next.提醒到 = `${ymd(now)} ${passed.at}`;
      const closing = afterQuiet(bridge.root, minutesOf(track.收工時間));
      if (minutes >= closing) next.收工過 = ymd(now);
      writeTodos(bridge.root, next);
      const upcoming = points.find((point) => point.real > minutes);
      const when = rest
        ? `今天放假（${rest}），不提醒；下個上班日 ${track.從幾點開始} 開始`
        : upcoming
        ? `下次提醒：今天 ${pad(Math.floor(upcoming.real / 60))}:${pad(upcoming.real % 60)}`
        : minutes < closing
          ? `今天的提醒時間都過了，${track.收工時間} 會給你收工清單`
          : `今天的時間都過了，明天 ${track.從幾點開始} 開始`;
      bridge.setReplyNote("待辦追蹤", `📝 待辦追蹤：已打開（${how}）；${when}`);
      return text(`已打開待辦追蹤：${how}。${when}。`);
    },
  });
}

/**
 * 每 30 秒由主程式呼叫一次（不排隊、不經模型）：到時間的提醒直接傳給使用者。
 * 一次性的送出後刪掉；循環的算出下一次。錯過好幾次的循環提醒只補發一次。
 */
export async function tick(bridge) {
  await trackTodos(bridge);
  const file = join(bridge.root, FILE);
  if (!existsSync(file)) return;
  let data;
  try {
    data = JSON.parse(readFileSync(file, "utf8"));
    if (!Array.isArray(data.提醒)) return;
  } catch {
    return; // 格式壞掉：工具那邊會告訴使用者；這裡不覆寫
  }
  const now = new Date();
  const due = data.提醒.filter((item) => item && typeof item.內容 === "string" && new Date(item.下次) <= now);
  if (due.length === 0) return;
  const keep = [];
  const messages = [];
  for (const item of data.提醒) {
    if (!due.includes(item)) {
      keep.push(item);
      continue;
    }
    const planned = new Date(item.下次);
    const late = now - planned > LATE_MINUTES * 60_000;
    messages.push([
      `⏰ 提醒：${item.內容}`,
      item.重複 ? `（${item.重複}）` : "",
      late ? `\n（原定 ${label(planned)}，那時秘書沒有在執行，現在補發）` : "",
    ].join(""));
    if (item.重複 && REPEAT.test(item.重複)) keep.push({ ...item, 下次: local(nextRepeat(item.重複, now)) });
  }
  // 先存檔再送：就算送的途中當掉，也不會重複提醒（送不出去的由主程式排進 outbox 補送）
  // 記下一個編號：送出刪掉的編號不會再被用到
  const next = Math.max(Number(data.下一個編號) || 0, ...data.提醒.map((item) => (Number(item?.編號) || 0) + 1));
  writeFileSync(file, `${JSON.stringify({ ...data, 下一個編號: next, 提醒: keep }, null, 2)}\n`);
  for (const text of messages) await bridge.notify(text);
}

/**
 * 待辦追蹤（tick 的一部分）：
 * - 每天第一個時間點之前，從最新一份工作日誌的「還沒完成的事」重新列今天的清單。
 * - 到時間點時，如果還有「待辦」狀態的，傳一則「還沒完成」；落在安靜時段就延到時段結束。
 * - 收工時間給一份清單（完成幾件、擱置幾件、還剩哪些）。沒開追蹤就什麼都不做。
 */
async function trackTodos(bridge) {
  const data = readTodos(bridge.root);
  if (data === null || !data.追蹤.開啟) return;
  const now = new Date();
  // 放假不追待辦：週末、國定假日（含補假）都不提醒，也不做收工清單。
  // 假日也想被追的話，把 待辦.json 的「只在工作日」改成 false。
  if (data.追蹤.只在工作日 !== false && (restDay(bridge.root, now) !== null)) return;
  const today = ymd(now);
  const minutes = now.getHours() * 60 + now.getMinutes();
  let changed = false;
  let current = data;
  if (data.日期 !== today) {
    // 新的一天：清單重新來過（擱置、完成都只算今天）
    const { day, items } = bridge.journalTodos();
    current = {
      ...data,
      日期: today,
      來源: day ? `日誌/${day}.md 的「還沒完成的事」` : "（還沒有工作日誌）",
      待辦: items.map((內容) => ({ 內容, 狀態: "待辦" })),
      提醒到: null,
      收工過: null,
    };
    changed = true;
  }
  const left = current.待辦.filter((item) => (item.狀態 ?? "待辦") === "待辦");
  const done = current.待辦.filter((item) => item.狀態 === "已完成");
  const held = current.待辦.filter((item) => item.狀態 === "擱置");
  // 這個時間點之前，最近一個該提醒的（延後過安靜時段）
  const due = checkpoints(current.追蹤)
    .map((at) => ({ at, real: afterQuiet(bridge.root, at) }))
    .filter((point) => minutes >= point.real)
    .at(-1);
  if (due && current.提醒到 !== `${today} ${due.at}` && left.length > 0) {
    current = { ...current, 提醒到: `${today} ${due.at}` };
    changed = true;
    await bridge.notify([
      `📝 還沒完成（${pad(Math.floor(due.at / 60))}:${pad(due.at % 60)}，${left.length} 件）`,
      todoLines(left),
      "",
      "做完或要擱置，跟我說一聲，例如「第 1 件做完了」「第 2 件今天先擱著」。",
    ].join("\n"));
  } else if (due && current.提醒到 !== `${today} ${due.at}`) {
    current = { ...current, 提醒到: `${today} ${due.at}` }; // 全部回報完了：記下來，不用提醒
    changed = true;
  }
  const endAt = afterQuiet(bridge.root, minutesOf(current.追蹤.收工時間));
  if (minutes >= endAt && current.收工過 !== today && current.待辦.length > 0) {
    current = { ...current, 收工過: today };
    changed = true;
    await bridge.notify([
      `🌆 今天收工了（${current.追蹤.收工時間}）`,
      `✅ 完成 ${done.length} 件${done.length ? `：${done.map((item) => item.內容).join("、")}` : ""}`,
      ...(held.length ? [`⏸ 擱置 ${held.length} 件：${held.map((item) => item.內容).join("、")}`] : []),
      left.length ? `📝 還沒完成 ${left.length} 件：\n${todoLines(left)}\n\n沒完成的明天晨報還會出現。` : "今天的待辦都回報完了，辛苦了。",
    ].join("\n"));
  }
  if (changed) writeTodos(bridge.root, current);
}
```

```js
/** 執行一個程式，收集輸出；超過 timeoutMs 就結束它。 */
function run(command, args, { cwd, env, timeoutMs = 60_000, verbatim = false } = {}) {
  return new Promise((resolve) => {
    const child = spawn(command, args, { cwd, env, stdio: ["ignore", "pipe", "pipe"], windowsHide: true, windowsVerbatimArguments: verbatim });
    children.add(child);
    let out = "";
    child.stdout.on("data", (chunk) => { out += chunk; });
    child.stderr.on("data", (chunk) => { out += chunk; });
    let timedOut = false;
    const timer = setTimeout(() => { timedOut = true; child.kill(); }, timeoutMs);
    child.on("close", (code) => {
      clearTimeout(timer);
      children.delete(child);
      resolve({ code, out: out.replace(/\r\n/g, "\n"), timedOut });
    });
  });
}
```

```js
const cleanUserFiles = (dir) => { for (const file of ["提醒.json", "排程.json", "通知.json", "固定會議.json", "情報.json"]) rmSync(join(dir, file), { force: true }); };

/** 在副本裡跑 unit(name)：固定時鐘、匯入提醒外掛，印出 JSON 結果。 */
async function unit(t, name) {
  const dir = t.makeCopy();
  cleanUserFiles(dir);
  try {
    const code = [
      `const RealDate = Date; let offset = new RealDate(2026, 9, 3, 21, 30).getTime() - RealDate.now();`,
      `class FakeDate extends RealDate { constructor(...a) { a.length ? super(...a) : super(RealDate.now() + offset); } static now() { return RealDate.now() + offset; } }`,
      `globalThis.Date = FakeDate; globalThis.__setNow = (...a) => { offset = new RealDate(...a).getTime() - RealDate.now(); };`,
      `const { units } = await import(${JSON.stringify(pathToFileURL(join(dir, "品檢", "L4-1.mjs")).href)});`,
      `console.log("結果:" + JSON.stringify(await units[${JSON.stringify(name)}](${JSON.stringify(dir)})));`,
    ].join("\n");
    const { out } = await t.run(process.execPath, ["--input-type=module", "-e", code], { cwd: dir, env: t.fakeEnv(), timeoutMs: 60_000 });
    const line = out.split("\n").find((text) => text.startsWith("結果:"));
    if (!line) throw new Error(`子程式沒有結果：\n${tail(out)}`);
    return JSON.parse(line.slice(3));
  } finally {
    t.removeCopy(dir);
  }
}

/** 用 out 裡的 [檢查] 行判斷：全部 ✅ 才合格。 */
function verdict(results) {
  const bad = results.filter((item) => !item.ok);
  const text = results.map((item) => `${item.ok ? "✅" : "❌"} ${item.what}${item.ok ? "" : `：${item.got}`}`).join("\n");
  return bad.length === 0 ? pass(text) : fail(text);
}

async function loadPlugin(dir) {
  const module = await import(pathToFileURL(join(dir, PLUGIN)).href);
  const tools = new Map();
  const notes = new Map();
  const sent = [];
  const bridge = {
    root: dir,
    setReplyNote: (key, text) => notes.set(key, text),
    notify: async (text) => { sent.push(text); },
    journalTodos: () => journalTodos,
  };
  module.default({ registerTool: (tool) => tools.set(tool.name, tool) }, bridge);
  const call = async (name, input) => {
    try {
      const result = await tools.get(name).execute("id", input);
      return { ok: true, text: result.content[0].text };
    } catch (error) {
      return { ok: false, text: error.message };
    }
  };
  const file = () => JSON.parse(readFileSync(join(dir, "提醒.json"), "utf8")).提醒;
  return { module, tools, notes, sent, bridge, call, file };
}

/** 從副本的 bridge.mjs 取出 nowLabel、pastWeekdayHint 兩個函式（bridge.mjs 一匯入就會啟動，不能直接 import）。 */
function weekFunctions(dir) {
  const code = readFileSync(join(dir, "bridge.mjs"), "utf8");
  const nowLabel = code.match(/function nowLabel\([\s\S]*?\n}\n/)?.[0];
  const hint = code.match(/function pastWeekdayHint\([\s\S]*?\n}\n/)?.[0];
  if (!nowLabel || !hint) return null;
  return new Function(`
    const pad = (value) => String(value).padStart(2, "0");
    const formatDate = (date) => \`\${date.getFullYear()}-\${pad(date.getMonth() + 1)}-\${pad(date.getDate())}\`;
    ${nowLabel}
    ${hint}
    return { nowLabel, pastWeekdayHint };`)();
}
```
