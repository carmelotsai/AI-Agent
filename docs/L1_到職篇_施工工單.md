你是這張工單的施工者。委託人是一般使用者，**不懂程式**。

- **先讀完整張工單，才可以動工。**
- 這張工單同時是**規格書**和**參考件**。第 3 章的程式碼都經過真機驗收，照抄就能動。
- 如果 Pi 已經出了新版，以第 2.2 節「進料檢驗」為準，對新版核對後再調整。**每一處跟參考件不同的地方，都要寫進施工日誌。**

### 3.6 工序六：主體 `bridge.mjs`

以下為真機驗收通過的參考件。**不可以更動的行為**：啟動時讀入人事資料組成 system prompt、白名單守門、訊息排隊一次處理一則、對話歷史只存在記憶體（重啟歸零）、通訊錯誤不中斷主程式、**回覆送不出去不能丟**、**Gemini 卡住要能自己重連**。

**網路設定（V1.2、V1.3）**：

- **連線等待 1 秒**：檔案開頭的 `net.setDefaultAutoSelectFamilyAttemptTimeout(1000)`。Node 建立連線時 IPv6、IPv4 各只等 250 毫秒，從台灣連 Telegram（伺服器在歐洲）延遲約 250～300 毫秒，常常還沒連上就放棄，黑視窗一直出現「輪詢錯誤：fetch failed」。實測每次都建新連線各 15 次：預設 9 次失敗，改成 1 秒後 0 次。
- **回覆送不出去先存著**：`send()` 一次送一則；失敗 2 秒後重試，還是失敗就存進 `outbox`，開頭加「（剛才網路斷線，這是補送的回覆）」，等收訊恢復（`getUpdates` 成功）後由 `flushOutbox()` 照順序補送。V1.1 以前送不出去的回覆會直接不見，委託人以為秘書「停住不動」。
- **Gemini 90 秒沒回應就重連（V1.3）**：檔案開頭把 `globalThis.fetch` 包一層，只管送到 `generativelanguage.googleapis.com` 的請求：90 秒內連開頭都沒回，或串流中途超過 90 秒沒有新資料，就主動中斷；錯誤訊息含「timeout」，Pi 會照 `.pi/settings.json` 的 `retry` 等 5 秒後自動重試。Node 內建 fetch 預設要等 5 分鐘才放棄，真機曾在網路不穩時整整卡了 5 分鐘才回。`/stop`（L2-5）的中斷訊號照樣轉過去。
- **處理中顯示「輸入中…」（V1.3）**：`showTyping()` 在收到訊息時送 `sendChatAction`（typing），每 4 秒一次，處理完就停。委託人看得出秘書在忙，不會以為他沒反應而一直追問。

globalThis.fetch = async (input, init = {}) => {
  const url = String(input instanceof Request ? input.url : input);
  if (!url.includes("generativelanguage.googleapis.com")) return plainFetch(input, init);
  const controller = new AbortController();
  const caller = init.signal; // /stop 會從這裡中斷，照樣轉過去
  if (caller?.aborted) controller.abort(caller.reason);
  caller?.addEventListener("abort", () => controller.abort(caller.reason), { once: true });
  let timer;
  const watch = () => {
    clearTimeout(timer);
    timer = setTimeout(
      () => controller.abort(new Error(`Gemini ${MODEL_IDLE_SECONDS} 秒沒有回應（timeout），重新連線`)),
      MODEL_IDLE_SECONDS * 1000,
    );
  };
  watch();
  try {
    const response = await plainFetch(input, { ...init, signal: controller.signal });
    if (!response.body) {
      clearTimeout(timer);
      return response;
    }
    watch(); // 回應是一段一段串流回來的；每收到一段就重新計時
    const body = response.body.pipeThrough(new TransformStream({
      transform(chunk, stream) {
        watch();
        stream.enqueue(chunk);
      },
      flush() {
        clearTimeout(timer);
      },
    }));
    return new Response(body, { status: response.status, statusText: response.statusText, headers: response.headers });
  } catch (error) {
    clearTimeout(timer);
    throw error;
  }
};

async function tg(method, body) {
  const response = await fetch(`${API}/${method}`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
    signal: AbortSignal.timeout(45_000),
  });
  const raw = await response.text();
  let data;

    data = JSON.parse(raw);
  } catch {
    throw new Error(`Telegram 回應不是 JSON（HTTP ${response.status}）`);
  }
  if (!response.ok || !data.ok) {
    throw new Error(data.description || `Telegram API 錯誤（HTTP ${response.status}）`);
  }
  return data;
}

/** 秘書處理訊息時，Telegram 顯示「輸入中…」（約 5 秒後自動消失，所以每 4 秒送一次）。回傳停止用的函式。 */
function showTyping(chatId) {
  const tick = () => {
    tg("sendChatAction", { chat_id: chatId, action: "typing" }).catch(() => {});
  };
  tick();
  const timer = setInterval(tick, 4000);
  return () => clearInterval(timer);
}

const outbox = []; // 送不出去的回覆，網路恢復後照順序補送

let sending = Promise.resolve();
/** 送出回覆；一次送一則，照呼叫的順序（網路不穩時才不會前後顛倒）。 */
function send(chatId, text) {
  const run = sending.then(() => sendNow(chatId, text));
  sending = run.catch(() => {});
  return run;
}

/** 失敗時 2 秒後再試一次，還是失敗就存進 outbox，網路恢復後補送（回覆不能掉）。 */
async function sendNow(chatId, text) {
  const body = { chat_id: chatId, text: text.slice(0, 4000) };
  if (outbox.length > 0) {
    outbox.push(body); // 前面還有沒送出的，排在後面才不會亂序
    return;
  }
  try {
    await tg("sendMessage", body);
  } catch (error) {
    console.error("回覆送出失敗，2 秒後重試：", errorText(error));
    await new Promise((resolve) => setTimeout(resolve, 2000));
    try {
      await tg("sendMessage", body);
    } catch (again) {
      console.error("回覆還是送不出去，網路恢復後補送：", errorText(again));
      outbox.push({ ...body, text: `（剛才網路斷線，這是補送的回覆）\n\n${body.text}`.slice(0, 4000) });
    }
  }
}

/** 收訊正常時呼叫：把 outbox 裡的回覆照順序補送；又失敗就留到下次。 */
function flushOutbox() {
  const run = sending.then(flushNow);
  sending = run.catch(() => {});
  return run;
}

async function flushNow() {
  while (outbox.length > 0) {
    try {
      await tg("sendMessage", outbox[0]);
    } catch {
      return;
    }
    outbox.shift();
    console.log("已補送一則回覆");
  }
}

let offset = 0;
while (true) {

      offset,

    await flushOutbox(); // 收得到訊息，代表網路恢復了
    for (const update of data.result ?? []) {

    console.error("輪詢錯誤：", errorText(error));

  }
}

/** 執行一個程式，收集輸出；超過 timeoutMs 就結束它。 */
function run(command, args, { cwd, env, timeoutMs = 60_000, verbatim = false } = {}) {
  return new Promise((resolve) => {
    const child = spawn(command, args, { cwd, env, stdio: ["ignore", "pipe", "pipe"], windowsHide: true, windowsVerbatimArguments: verbatim });
    children.add(child);
    let out = "";
    child.stdout.on("data", (chunk) => { out += chunk; });
    child.stderr.on("data", (chunk) => { out += chunk; });
    let timedOut = false;
    const timer = setTimeout(() => { timedOut = true; child.kill(); }, timeoutMs);
    child.on("close", (code) => {
      clearTimeout(timer);
      children.delete(child);
      resolve({ code, out: out.replace(/\r\n/g, "\n"), timedOut });
    });
  });
}
